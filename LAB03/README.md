# LAB03

# Structure

RAG-Project/
│
├── data/                                   # Stores the original documents
│   └── cat_qa_dataset.txt                  # Sample questions for testing
│
├── outputs/                                # Stores outputs generated by each lab
│   ├── extracted_text.json                 # Extracted PDF text with page numbers
│   ├── chunks.json                         # Text chunks with metadata
│   ├── embeddings.npy                      # Embedding vectors
│   └── retrieval_results.json              # Top-k retrieval results
│
├── vector_db/                              # Stores the vector database
│   ├── document.index                      # FAISS vector index
│   └── chunk_store.json                    # Chunks and metadata
│
├── labs/                                   # Step-by-step lab scripts
│   ├── lab01_extract_text.py               # Extract text from PDF
│   ├── lab02_chunking.py                   # Split text into chunks
│   ├── lab03_create_embeddings.py          # Generate embeddings
│   ├── lab04_create_vector_db.py           # Build the FAISS vector database
│   ├── lab05_query_embedding.py            # Create query embeddings
│   ├── lab06_similarity_search.py          # Retrieve Top-k relevant chunks
│   └── lab07_complete_retrieval.py         # Complete retrieval pipeline
│
├── src/                                    # Reusable core modules
│   ├── __init__.py                         # Python package initialization
│   ├── document_loader.py                  # PDF loading and text extraction
│   ├── text_splitter.py                    # Text chunking
│   ├── embedding_model.py                  # Embedding model
│   ├── vector_store.py                     # FAISS vector database
│   └── retriever.py                        # Semantic retrieval
│
├── config.py                               # Project configuration
├── requirements.txt                        # Required Python packages
└── main.py                                 # Run the retrieval system
